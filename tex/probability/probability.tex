\chapter{Probability}

\section{Basics}

\begin{definition}
  The \emph{sample space}, denoted $\Omega$ is the set of all possible outcomes of an experiment.

  Each element $\omega \in \Omega$ is called an \emph{outcome}. An \emph{event} $E$ is a subset of $\Omega$, and is hence a set of possible outcomes. Thus $\Omega$ itself is an event, the \emph{certain event}, and so is the empty set $\O$, the \emph{impossible event}. An event $E$ \emph{occurs} if the outcome $\omega$ is in $E$.
\end{definition}

\begin{definition}
  If $\Omega$ is countable, then we say that it is a \emph{discrete} sample space. Otherwise, if $\Omega$ is uncountable, we say that it is a \emph{continuous} sample space.
\end{definition}

Since events are sets, we can use the usual properties of sets. In particular, if $A$ and $B$ are events, then $A \cup B$ is an event (that either $A$ or $B$ occur) and $A \cap B$ is an event (that both $A$ and $B$ occur).

\begin{proposition}[Set theory properties]
  Let $A$, $B$, $C$ be subsets of $\Omega$.
  \begin{enumerate}
\item We have $A \cup \O = A$ and $A \cap \O = \O$.
    \item We have $A \cup \Omega = \Omega$ and $A \cap \Omega = A$. 
    \item We have $A \cup (\Omega \setminus A) = \Omega$ and $A \cap (\Omega \setminus A) = \O$. In other words, $A \cup A^{c} =\Omega$ and $A \cap A^{c} = \O$.
    \item (Commutativity) We have $A \cup B = B \cup A$  and $A \cap B = B \cap A$.
    \item (Associativity) We have $A \cup (B \cup C) = (A \cup B) \cup C$ and $A \cap (B \cap C) = (A \cap B) \cap C$.
    \item (Distributivity) We have $A \cup (B \cap C) = (A \cup B) \cap (A \cup C)$ and $A \cap (B \cup C) = (A \cap B) \cup (A \cap C)$.
    \item (De Morgan's laws) We have $\Omega \setminus (A \cup B) = (\Omega \setminus A) \cap (\Omega \setminus B)$ and $\Omega \setminus (A \cap B) = (\Omega \setminus A) \cup (\Omega \setminus B)$. In other words, $(A \cup B)^{c} = A^{c} \cap B^{c}$ and $(A \cap B)^{c} = A^{c} \cup B^{c}$.
  \end{enumerate}
\end{proposition}

\begin{definition}
  Two events $A$ and $B$ are \emph{mutually exclusive} or \emph{disjoint} if $A \cap B = \O$. A collection of events $\{E_i\}_{i \in I}$ is mutually exclusive if $E_i \cap E_j = \O$ for all $i \neq j$.
\end{definition}

\begin{definition}
  Let $\Omega$ be a sample space and let $2^{\Omega}$ denote the power set of $\Omega$. Then  $\prob \colon 2^{\Omega} \to [0, 1]$ is a probability function that assigns to each event a probability. This function $\prob$ must satisfy the following axioms formulated by Kolmogorov:
  \begin{enumerate}
    \item For all events $E$, we have $\prob(E) \ge 0$.
    \item The probability of the sample space is $1$. That is, $\prob(\Omega) = 1$.
    \item (Countable additivity) Let $E_1, E_2, \dots$ be a countable sequence of mutually exclusive (disjoint) events. Then, \[
    \prob\left(\, \bigcup_{i = 1}^{\infty} E_i \right) = \sum_{i = 1}^{\infty} \prob(E_i).
    \] 
  \end{enumerate}
\end{definition}

\begin{remark}
  This differs from the definition of probability space one will encounter later on in a more advanced, measure theoretic course. The formal definition of a probability space is the following: a probability space is a tuple $(\Omega, \mathcal{F}, \prob)$, where $\Omega$ is the sample space, $\mathcal{F} \subseteq 2^{\Omega}$ is a $\sigma$-algebra, and $\prob$ is a probability measure $\prob \colon \mathcal{F} \to [0, 1]$ satisfying $\prob(A) \ge 0$, $\prob(\Omega) = 1$, and countable additivity.

  In particular, our definition does not work for some uncountable sample spaces. The restriction for $\mathcal{F}$ to be a $\sigma$-algebra is exactly what is needed to `fix' this issue.
\end{remark}

\begin{proposition}[Properties of probability] \label{prop:properties}
  Let $\Omega$ be a sample space and let $A, B$ be events. Let $\prob$ be a probability function on $\Omega$.
  \begin{enumerate}
    \item We have $\prob(\O) = 0$.
    \item We have $\prob(A^{c}) = \prob(\Omega \setminus A) = 1 - \prob(A)$.
    \item We have $\prob(A) \le 1$.
    \item We have $\prob(A \cup B) = \prob(A) + \prob(B) - \prob(A \cap B)$.
    \item If $A \subseteq B$, then $\prob(A) \le \prob(B)$.
    \item (Finite additivity) If $E_1, E_2, \dots, E_n$ are mutually exclusive events, then \[
      \prob \left(\, \bigcup_{i = 1}^{n} E_i \right) = \sum_{i = 1}^{n} \prob(E_i)\]
    \item If $A_1 \subseteq A_2 \subseteq \dots$ and $B = \bigcup_{i = 1}^{\infty} A_i$, then $\prob(B) = \lim_{n \to \infty} \prob(A_n)$.
    \item If $A_1 \supseteq A_2 \supseteq \dots$ and $B = \bigcap_{i = 1}^{\infty} A_i$, then $\prob(B) = \lim_{n \to \infty} \prob(A_n)$.
  \end{enumerate}
\end{proposition}
\begin{proof}
  We will prove (1), (2), (4), (6), and (7), and leave the rest as exercises.
  \begin{enumerate}
    \item This is a corollary of (2). Since $\O = \Omega^{c}$, we have $\prob(\O) = \prob(\Omega^{c}) = 1 - \prob(\Omega) = 1 - 1 = 0$.
    \item This is a corollary of finite additivity (6), with $n = 2$. Consider the disjoint events $A$ and $A^{c}$. Since $A \cup A^{c} = \Omega$, we have $1 = \prob(\Omega) = \prob(A \cup A^{c}) = \prob(A) + \prob(A^{c})$, from which the result follows.
    \item Exercise.
    \item Again, this follows from finite additivity applied to the disjoint sets $A$ and $B \setminus A$. Since $A \cup (B \setminus A) = A \cup B$, we have 
      \begin{align*}
        \prob(A \cup B) &= \prob(A \cup (B \setminus A)) \\
                        &= \prob(A) + \prob(B \setminus A).
      \end{align*}
      Notice that $B = (B \setminus A) \cup (A \cup B)$ is a disjoint union, from which we get $\prob(B) = \prob(B \setminus A) + \prob(A \cap B)$, so $\prob(B \setminus A) = \prob(B) - \prob(A \cap B)$. This, combined with the previous equation, yields \[
        \prob(A \cup B) = \prob(A) + \prob(B) - \prob(A \cap B).
      \] 
    \item Exercise.
    \item Consider the mutually exclusive events $E_1, \dots, E_n$. Let $E_{n+1} = E_{n+2} = \dots = \O$. Notice that $E_1, \dots$ still form a mutually exclusive collection of events, so countable additivity applies. Thus 
      \begin{align*}
        \prob \left( \, \bigcup_{i = 1}^{\infty} E_i \right) &= \sum_{i = 1}^{\infty} \prob(E_i) \\
                                                             &= \sum_{i = 1}^{n} \prob(E_i) + \sum_{i = n+1}^{\infty} \prob(E_i) \\
                                                             &= \sum_{i = 1}^{n} \prob(E_i) + 0,
      \end{align*}
      since $\prob(E_i) = \prob(\O) = 0$ for all $i > n$. Hence finite additivity holds.
    \item Define $B_i = A_i \setminus A_{i - 1}$. Clearly all of $B_i$ are mutually exclusive, and moreover, $B = \bigcup_{i = 1}^{\infty} B_i$. Hence by countable additivity, 
      \begin{align*}
        \prob(B) &= \prob \left( \, \bigcup_{i = 1}^{\infty} B_i \right) \\
                 &= \sum_{i = 1}^{\infty} \prob(B_i) \\
                 &= \lim_{n \to \infty} \sum_{i = 1}^{n} \prob(B_i) \\
                 &= \lim_{n \to \infty} \prob \left( \, \bigcup_{i = 1}^{n} B_i \right),
      \end{align*}
      with the last step being due to finite additivity. But $\bigcup_{i = 1}^{n} B_i = A_n$, so we have \[
        \prob(B) = \lim_{n \to \infty} \prob(A_n)
      \] as required.
    \item Exercise. \qedhere
  \end{enumerate}
\end{proof}

In a discrete sample space, all outcomes are disjoint. (This is obviously true, due to the fact that there can be only one outcome of an experiment.) Hence for any event $E$, we have that \[
  \prob(E) = \sum_{\omega \in E} \prob(\{\omega\}).
\] 
\begin{remark}
  It is important to note that the statements $\prob(E) = 0 \implies E = \O$ and $\prob(E) = 1 \implies E = \Omega$ are \warning{not necessarily true}. For example, let $\Omega = \{H, T\}$ and define $\prob(H) = 0$ and $\prob(T) = 1$. Notice that $H \neq \O$ and $T \neq \Omega$.
\end{remark}

\section{Conditional probability and independence}

\begin{definition}
  The \emph{conditional probability} of event $A$ \emph{given} event $B$ is defined as \[
    \prob(A \mid B) = \frac{\prob(A \cap B)}{\prob(B)},
  \] assuming that $\prob(B) > 0$.
\end{definition}

This definition may be rearranged to get the equivalent formulation \[
  \prob(A \cap B) = \prob(A \mid B) \prob(B),
\] which sometimes may be more useful.

\begin{definition}
  We say that events $A$ and $B$ are \emph{positively related} (or \emph{positively correlated}) if $\prob(A \mid B) > \prob(A)$. Similarly, events  $A$ and $B$ are \emph{negatively related} if $\prob(A \mid B) < \prob(A)$.
\end{definition}

A simple calculation will show that if $\prob(A \mid B) > \prob(A)$, then $\prob(B \mid A) > \prob(B)$. Hence the intuition is that if two events are positively related, then knowing one event will boost the likelihood of the other event.

\begin{definition}
  Two events $A$ and $B$ are said to be \emph{independent} if \[
    \prob(A \cap B) = \prob(A) \prob(B).
  \] 
\end{definition}

If $A$ and $B$ are independent, then $\prob(A \mid B) = \prob(A)$, and similarly, $\prob(B \mid A) = \prob(B)$. However, these equations involving conditional probability only make sense when $\prob(A) > 0$ and $\prob(B) > 0$. Hence we have defined independence to make sense even when some of the probabilities are zero.

\begin{proposition}[Properties of independence]
  Suppose that $A$ and $B$ are independent events.
  \begin{enumerate}
    \item We have that $A$ and $B^{c}$ are independent.
    \item We have that $A^{c}$ and $B$ are independent.
    \item We have that $A^{c}$ and $B^{c}$ are independent.
  \end{enumerate}
\end{proposition}
\begin{proof} \leavevmode
  \begin{enumerate}
    \item Since $A$ and $B$ are independent, we have $\prob(A \cap B) = \prob(A) \prob(B)$. Since $A = (A \cap B) \cup (A \cap B^{c})$ is a disjoint union, we have
      \begin{align*}
        \prob(A) &= \prob(A \cap B) + \prob(A \cap B^{c}) \\
        \prob(A \cap B^{c}) &= \prob(A) - \prob(A \cap B) \\
                            &= \prob(A) - \prob(A) \prob(B) \\
                            &= \prob(A) (1 - \prob(B)) \\
                            &= \prob(A) \prob(B^{c}),
      \end{align*}
      so we see that $A$ and $B^{c}$ are independent.
    \item Follows directly from (1) by swapping $A$ and $B$.
    \item By de Morgan, we have $A^c \cap B^c = (A \cup B)^{c}$. Hence 
      \begin{align*}
        \prob(A^c \cap B^c) &= 1 - \prob(A \cup B) \\
                            &= 1 - (\prob(A) + \prob(B) - \prob(A \cap B)) \\
                            &= 1 - \prob(A) - \prob(B) + \prob(A) \prob(B) \\
                            &= (1 - \prob(A)) (1 - \prob(B)) \\
                            &= \prob(A^{c})\prob(B^{c}),
      \end{align*}
      so $A^{c}$ and $B^{c}$ are independent. \qedhere
  \end{enumerate}
\end{proof}

We can extend our definition of independence to multiple events.
\begin{definition}
  Events $E_1, E_2, \dots, E_n$ are \emph{independent} or \emph{mutually independent} if for every $k \le n$ and $1 \le i_1 \le\dots \le i_k \le n$, \[
    \prob \left( \, \bigcap_{j = 1}^{k} E_{i_j} \right) = \prod_{j = 1}^{k} \prob(E_{i_j}).
  \] 
\end{definition}
\begin{remark}
  This definition is different from that of \emph{pairwise} independence, which only requires that every pair of events is independent. Mutual independence is strictly stronger, in the sense that mutual independence implies pairwise independence, but not the converse.
\end{remark}

\begin{definition}
We say that a collection of events $\{E_i\}_{i \in I}$ is \emph{exhaustive} if \[
    \bigcup_{i \in I} E_i = \Omega.
  \] 
\end{definition}

In particular, note that $A$ and $A^{c}$ are exhaustive, and also disjoint.

\begin{definition}
  A \emph{partition} of the sample space $\Omega$ is a collection of events that are mutually exclusive and exhaustive. That is, if $\{E_i\}_{i \in I}$ is a collection of events, then it is a partition if and only if: 
  \begin{enumerate}
    \item (Mutually exclusive) For all $i \neq j$, $E_i \cap E_j = \O$.
    \item (Exhaustive) We have $\bigcup_{i \in I} E_i = \Omega$.
  \end{enumerate}
  A partition $\{E_i\}_{i \in I}$ is countable if the set is countable, and uncountable if the set is uncountable.
\end{definition}

\begin{proposition}[Law of total probability]
  If $\{E_i\}_{i \in I}$ is a countable partition of $\Omega$, then for any event $A$, \[
    \prob(A) = \sum_{i \in I} \prob(A \cap E_i).
  \] or equivalently, \[
    \prob(A) = \sum_{i \in I} \prob(A \mid E_i) \prob(E_i).
  \] 
\end{proposition}

\begin{proposition}[Bayes' formula]
  Given events $A$ and $B$, with $\prob(B) > 0$, we have \[
    \prob(A \mid B) = \frac{\prob(B \mid A) \prob(A)}{\prob(B)}.
  \] Equivalently, if $\{E_i\}_{i \in I}$ is a countable partition of $\Omega$, and $A$ is an event with $\prob(A) > 0$, then \[
    \prob(E_k \mid A) = \frac{\prob(A \mid E_k) \prob(E_k)}{\sum_{i \in I} \prob(A \mid E_i) \prob(E_i)}.
  \] 
\end{proposition}

\section{Random variables}

\begin{definition}
  A \emph{random variable} $X$ is a function $X \colon \Omega \to \R$ that assigns a real number to every outcome in the sample space.

  We define the probability of $X$ taking on a value $x$ as \[
    \prob(X = x) = \prob(\{\omega \in \Omega : X(\omega) = x\}).
  \] In general, for any subset $A \subseteq \R$, we have \[
    \prob(X \in A) = \prob(\{\omega \in \Omega : X(\omega) \in A\}).
  \] 
\end{definition}

\begin{definition}
  The \emph{state space} of $X$, the set of possible values that $X$ can take, is denoted $S_X$. If $S_X$ is countable, then $X$ is a \emph{discrete} random variable.
\end{definition}

\begin{remark}
  One can have a discrete random variable in an uncountable sample space. For example, let $\Omega = \R$ and define \[
    X = 
    \begin{cases*}
      1 & if $\omega \ge 0$, \\
      0 & otherwise.
    \end{cases*}
  \]
\end{remark}

\begin{remark}
  A random variable partitions $\Omega$. This is the point of having random variables \Emdash to carve up the sample space into smaller chunks, to simplify calculations.
\end{remark}

\begin{definition}
  The \emph{probability mass function} (or \emph{pmf}) of a discrete random variable $X$ is a function $p_X \colon S_X \to [0, 1]$ defined by \[
    p_X(x) = \prob(X = x).
  \] 

  The \emph{cumulative distribution function} or \emph{distribution function} or \emph{cdf} of any random variable $X$ is a function $F_X \colon \R \to [0, 1]$ defined by \[
    F_X(x) = \prob(X \le x).
  \] 
\end{definition}

\begin{proposition}
  The pmf $p_X$ of a discrete random variable $X$ satisfies:
  \begin{enumerate}
    \item For all $x \in S_x$, $p_X(x) \ge 0$.
    \item The sum over all $x$ in the state space is $\sum_{x \in S_X} p_X(x) = 1$.
  \end{enumerate}
\end{proposition}

\begin{proposition}[Properties of cdfs]
  The cdf $F_X$ of a random variable $X$ satisfies:
  \begin{enumerate}
    \item For real numbers $a < b$, we have $\prob(a < X \le b) = F_X(b) - F_X(a)$.
    \item $F_X$ is non-decreasing (increasing).
    \item $\lim_{x \to -\infty} F_X(x) = 0$ and $\lim_{x \to \infty} F_X(x) = 1$.
    \item $F_X$ is right-continuous. That is, $\lim_{x \to y^{+}} F_X(x) = F_X(y)$ for all $y \in \R$.
  \end{enumerate}
\end{proposition}

\begin{proposition}
  A function $G$ is the cdf of a random variable if and only if  $G$ is non-decreasing, right-continuous, and has $\lim_{x\to -\infty} G(x) = 0$ and $\lim_{x \to \infty} G(x) = 1$.
\end{proposition}

\begin{definition}
  A random variable $X$ is \emph{continuous} if its distribution function $F_X$ is continuous (in particular, it is left-continuous as well as right-continuous). In this case, $S_X$ will be uncountable.
\end{definition}

\begin{proposition}
  Let $X$ be a random variable. Then,  \[
    \prob(X < x) =  \lim_{t \to x^{-}} F_X(t)
  \] and \[
    \prob(X = x) = F_X(x) - \lim_{t \to x^{-}} F_X(t).
  \] In particular, $P(X = x) = 0$ for all $x \in S_X$ if and only if $F_X$ is continuous.
\end{proposition}
\begin{proof}
  The second equation $\prob(X = x) = F_X(x) - \lim_{t \to x^{-}}F_X(t)$ immediately follows from the first, due to the fact that $\prob(X = x) = \prob(X \le x) - \prob(X < x)$.

  Consider $\prob(X \le x - h) = F_X(x - h)$ for some real number $h > 0$. Then $\prob(X < x) = \lim_{h \to 0} \prob(X \le x - h) = \lim_{h\to 0} F_X(x-h) = \lim_{t \to x^{-}}F_X(t)$, as required. (Formally, we would use $A_i \coloneq \{X \le x - \frac{1}{n} \}$ and apply Proposition~\ref{prop:properties}(8).)

  Now we see that $\prob(X = x) = 0$ if and only if $F_X(x) = \lim_{t\to x^{-}}F_X(t)$, which is equivalent to left-continuity. But since all distribution functions are already right-continuous, adding left-continuity simply makes it continuous. Hence  $\prob(X = x) = 0$ for all $x$ if and only if $F_X$ is continuous.
\end{proof}

\begin{definition}
  A function $f \colon \R \to [0, \infty)$ is said to be a \emph{probability density function} or \emph{pdf} of a continuous random variable $X$ if \[
    \int_{-\infty}^{x} f(t) \dl t = F_X(x).
  \] If it exists, the pdf of a random variable is unique.
\end{definition}

\begin{proposition}[Properties of pdfs]
  Let $X$ be a continuous random variable with probability density function $f_X$. Then:
  \begin{enumerate}
    \item For all $x \in \R$, we have $f_X(x) \ge 0$.
    \item $\displaystyle \int_{-\infty}^{\infty} f_X(t) \dl t = 1$.
    \item For real numbers $a < b$, \[
        \prob(a < X \le b) = \int_{a}^{b} f_X(t) \dl t.
      \]
    \item For almost all values of $x$, \[
        \diff{F_X(x)}{x}  = f_X(x).
    \] 
  \end{enumerate}
\end{proposition}

\begin{remark}
  Importantly, \warning{not all} continuous random variables have a pdf. These are, however, rare and pathological.
\end{remark}

\section{Expectation}

\begin{definition}
  Let $X$ be a discrete random variable with pmf $p_X$. Then, the \emph{expected value}, or \emph{mean} of $X$ is defined to be \[
    \E[X] = \sum_{x \in S_X} xp_X(x),
  \] provided that the sum converges absolutely. If the state space $S_X$ is finite, then the expected value is always defined. The expected value is sometimes denoted $\mu_X$ or simply $\mu$ if $X$ is clear from context.
\end{definition}

\begin{definition}
  Let $X$ be a continuous random variable with pdf $f_X$. Then, the expected value of $X$ is defined to be \[
    \E[X] = \int_{-\infty}^{\infty} xf_X(x) \dl x,
  \] provided that the integral converges absolutely.
\end{definition}

\begin{proposition} \label{prop:lotus}
  Let $X$ be a random variable and let $g\colon \R \to \R$ be a function.
  \begin{enumerate}
    \item If $X$ is a discrete random variable with pmf $p_X$, then \[
        \E[g(X)] = \sum_{x \in S_X} g(x) p_X(x),
      \] provided the sum converges absolutely.
    \item If $X$ is a continuous random variable with pdf $f_X$, then \[
        \E[g(X)] = \int_{-\infty}^{\infty} g(x)f_X(x) \dl x,
      \] provided that the integral converges absolutely.
  \end{enumerate}
\end{proposition}
\begin{proof}
  We prove the discrete case, and leave the continuous case as an exercise. Clearly, $Y \coloneq g(X)$ is a discrete random variable, so working from the definition, we have
  \begin{align*}
    \E[g(X)] = \E[Y] &= \sum_{y \in S_Y} y \prob(Y = y) \\
                     &= \sum_{y \in S_Y} y \prob(g(X) = y).
  \end{align*}
  Let $g^{-1}(y) \coloneq \{x \in \R : g(x) = y\}$ be the preimage of $y$. Then, we see that $g(X) = y$ is equivalent to $X \in g^{-1}(y)$, so 
  \begin{align*}
    \E[Y] &= \sum_{y \in S_Y} y \prob(X \in g^{-1}(y)) \\
          &= \sum_{y \in S_Y} y \sum_{x \in g^{-1}(y)} \prob(X = x) \\
          &= \sum_{y \in S_Y} \sum_{x \in g^{-1}(y)} y \prob(X = x),
  \end{align*}
  where the swapping of the summations is justified by absolute convergence. Now, inside the sum, $y = g(x)$, and moreover,  $\prob(X = x) = p_X(x)$ by definition. Note that summing over all $y \in S_Y$ and $x \in g^{-1}(y)$ is equivalent to summing over all $x \in S_X$, since $S_X$ is partitioned by the set of all preimages of $y \in S_Y$. Thus, we obtain \[
    \E[g(X)] = \E[Y] = \sum_{x \in S_X} g(x) p_X(x)
  \] as desired.
\end{proof}

\begin{proposition}
  Let $X$ be a random variable, and $a, b \in \R$ be constants. Then, \[
    \E[aX + b] = a\E[X] + b.
  \] 
\end{proposition}

\begin{definition}
  The \emph{variance} of a random variable $X$ is defined by \[
    \var(X) = \E[(X - \E[X])^2].
  \] This is sometimes denoted as $\sigma^2_X$ or just  $\sigma^2$ if $X$ is clear from context. 
\end{definition}

\begin{definition}
  The \emph{standard deviation}, denoted $\sigma_X$ or $\sigma$ or $\sd(X)$, of a random variable $X$ is the square root of its variance. That is, \[
    \sigma_X = \sqrt{\var(X)} = \sqrt{\E[(X - \E[X])^2]}.
  \] 
\end{definition}

\begin{remark}
  The standard deviation is a measure of the spread or variability of a random variable. This is because the variation measures the average squared distance to the mean. The reason why the standard deviation is the square root of the mean is to make the units equal to the units of the original random variable. The reason why the variation is the expected value of the squared distance to the mean, instead of absolute value or some other metric, is because it simplifies calculations.
\end{remark}

\begin{proposition}[Properties of variance and standard deviation]
  Let $X$ be a random variable.
  \begin{enumerate}
    \item $\var(X) \ge 0$.
    \item If $Y = aX + b$ for real numbers $a$ and $b$, then $\var(Y) = a^2\var(X)$ and $\sd(Y) = \abs{a}\sd(X)$.
    \item $\var(X) = \E[X^2] - \E[X]^2$.
  \end{enumerate}
\end{proposition}

\begin{definition}
  The $k$\textsuperscript{th} \emph{moment} (about the origin) of a random variable $X$ is given by \[
    \mu_k = \E[X^k].
  \] The $k$\textsuperscript{th} \emph{central moment} of a random variable $X$ is given by \[
  \nu_k = \E[(X - \E[X])^{k}].
  \] Thus $\var(X) = \nu_2 = \mu_2 - \mu_1^2$.
\end{definition}

\begin{remark}
  The first few moments give valuable information about the distribution. 
  \begin{itemize}
    \item The first moment $\mu_1$ tells you the mean.
    \item The second central moment $\nu_2$ tells you the variance.
    \item The third central moment $\nu_3$ (after normalisation) tells you the \emph{skewness}, that is, how lopsided the distribution is. A random variable with a perfectly symmetric pdf will have a skewness of zero.
    \item The fourth central moment $\nu_4$ (after normalisation) tells you about the heaviness of the tail.
  \end{itemize}
\end{remark}

\begin{proposition}[Moment via tail probability formula]
  Let $X$ be a non-negative random variable and $n$ be a positive integer. Then, \[
    \E[X^{n}] = n \int_{0}^{\infty} x^{n-1} (1 - F_X(x)) \dl x.
  \] 
\end{proposition}
\begin{proof}
  First, note that \[
    x^{n} = \int_{0}^{x} ny^{n-1} \dl y.
  \] Thus, by Proposition~\ref{prop:lotus}, we have 
  \begin{align*}
    \E[X^n] &= \int_{0}^{\infty} x^n f_X (x) \dl x \\
            &= \int_{0}^{\infty} \left( \int_{0}^{x} ny^{n-1} \dl y \right) f_X(x) \dl x.
  \end{align*}
  We may apply Fubini's theorem (or rather, Tonelli's theorem due to the non-negativity of the integrand) to exchange the order of integration. The innermost integral will be integrated with respect to $x$ from $y$ to $\infty$, because we have that $y \le x$, and $x$ is unbounded. The outermost integral will be integrated with respect to $y$ from $0$ to $\infty$. Thus, we have 
  \begin{align*}
    \E[X^n] &= \int_{0}^{\infty} ny^{n-1} \left( \int_{y}^{\infty} f_X(x) \dl x \right) \dl y \\
            &= n\int_{0}^{\infty} y^{n-1} \prob(X > y) \dl y \\
            &= n \int_{0}^{\infty} y^{n-1} (1 - F_X(y)) \dl y,
  \end{align*}
  from which the formula is obtained after a changing the dummy variable $y$ back to $x$.
\end{proof}

\section{Discrete probability distributions}

\subsection{Bernoulli}

\begin{definition}
  A random variable $X$ has a \emph{Bernoulli distribution} with parameter $p \in [0, 1]$ if $S_X = \{0, 1\}$ and its pmf is
  \begin{align*}
    p_X(1) &= p \\
    p_X(0) &= 1 - p.
  \end{align*}
  We say that $X$ is a \emph{Bernoulli random variable}.

  A \emph{Bernoulli trial} is an experiment whose outcome is modeled by a Bernoulli random variable.
\end{definition}

\begin{proposition}
  If $X$ has a Bernoulli distribution with parameter $p$, then 
  \begin{align*}
    \E[X] &= p \\
    \var(X) &= p(1-p).
  \end{align*}
\end{proposition}

\subsection{Binomial}

\begin{remark}
  A binomial random variable is equivalent to the number of successes in $n$ independent Bernoulli trials.
\end{remark}

\subsection{Geometric}

\begin{remark}
  A geometric random variable can be thought of as the number of failures before the first success.
\end{remark}

\subsection{Negative binomial}

\begin{remark}
  A negative binomial random variable can be thought of as the number of failures before the $r$th success.
\end{remark}

\subsection{Hypergeometric}

\begin{remark}
  A hypergeometric random variable can thought of as the number of failures before the $r$th success, without replacement. For example, a hypergeometric random variable describes the number of blue balls taken from a basket of blue and red balls before taking the $r$th red ball.
\end{remark}
